1) Struttura narrativa dell’animazione: Medusa diventa “sorgente viva”
Come devono comportarsi i colpi
Non partono più dal container rosso, ma dagli occhi di Medusa.

Ogni colpo è serpentino, come un cobra che si lancia e poi ritorna.

Ogni colpo colpisce un forziere bancario (simbolo della vulnerabilità umana).

L’impatto genera una piramide di denaro nel “Regno delle Tre Meduse”.

Il ritorno del colpo verso Medusa crea un ciclo ipnotico:
percezione → attacco → rivelazione → ricostruzione → consapevolezza.

Ruolo del container rosso
Il container rosso diventa:

un monitor delle minacce, non la sorgente dei colpi

un elemento che “traduce” in parole ciò che Medusa percepisce

un pannello che si espande leggermente ad ogni colpo, come se “registrasse” l’attacco

javascript
// ===============================
//  MEDUSA ENGINE – CORE MODULE
// ===============================

// Carica il JSON (puoi anche incorporarlo direttamente qui)
async function loadMedusaData() {
    const response = await fetch("./data/medusa-data.json");
    return await response.json();
}

// ===============================
//  ANIMAZIONE: COLPI DAGLI OCCHI
// ===============================

function launchSerpentStrike(text) {
    const medusaEyes = document.querySelectorAll(".medusa-eye");
    const battlefield = document.querySelector("#battlefield");

    medusaEyes.forEach(eye => {
        const strike = document.createElement("div");
        strike.className = "serpent-strike";
        strike.innerText = text;

        const eyeRect = eye.getBoundingClientRect();
        strike.style.left = eyeRect.left + eyeRect.width / 2 + "px";
        strike.style.top = eyeRect.top + eyeRect.height / 2 + "px";

        battlefield.appendChild(strike);

        // Movimento serpentino
        strike.animate([
            { transform: "translate(0,0) rotate(0deg)" },
            { transform: "translate(80px,-40px) rotate(20deg)" },
            { transform: "translate(160px,20px) rotate(-10deg)" },
            { transform: "translate(240px,-60px) rotate(15deg)" },
            { transform: "translate(300px,0) rotate(0deg)" }
        ], {
            duration: 1800,
            easing: "ease-in-out"
        });

        // Impatto sul forziere
        setTimeout(() => {
            createMoneyPyramid();
            strike.remove();
        }, 1800);
    });
}

// ===============================
//  PIRAMIDE DI DENARO
// ===============================

function createMoneyPyramid() {
    const zone = document.querySelector("#medusa-realm");
    const pyramid = document.createElement("div");
    pyramid.className = "money-pyramid";

    pyramid.innerHTML = "💰";

    zone.appendChild(pyramid);

    pyramid.animate([
        { transform: "scale(0.2)", opacity: 0 },
        { transform: "scale(1.2)", opacity: 1 },
        { transform: "scale(1)", opacity: 1 }
    ], {
        duration: 900,
        easing: "ease-out"
    });
}

// ===============================
//  AGGIORNAMENTO CONTAINER ROSSO
// ===============================

function updateThreatContainer(text) {
    const container = document.querySelector("#threat-container");
    container.innerText = text;

    container.animate([
        { transform: "scale(1)" },
        { transform: "scale(1.08)" },
        { transform: "scale(1)" }
    ], {
        duration: 600,
        easing: "ease-out"
    });
}

// ===============================
//  CICLO PRINCIPALE DELLE TRUFFE
// ===============================

async function startMedusaCycle() {
    const data = await loadMedusaData();
    const categories = data.truffe;

    let allThreats = [];
    categories.forEach(cat => allThreats.push(...cat.voci));

    let index = 0;

    setInterval(() => {
        const threat = allThreats[index];
        updateThreatContainer(threat);
        launchSerpentStrike(threat);

        index = (index + 1) % allThreats.length;
    }, 3500);
}

// ===============================
//  ZONA DI COMFORT – BANDIERE
// ===============================

async function populateComfortZone() {
    const data = await loadMedusaData();
    const banks = data.centralBanks;

    const zone = document.querySelector("#comfort-zone");

    banks.forEach(bank => {
        const btn = document.createElement("a");
        btn.className = "flag-button";
        btn.href = bank.url;
        btn.target = "_blank";

        btn.innerHTML = `
            <img src="./Images/flags/${bank.flag}" class="flag-icon">
            <span>${bank.name}</span>
        `;

        zone.appendChild(btn);
    });
}

// ===============================
//  RIPOSIZIONAMENTO REPLAY BUTTON
// ===============================

function repositionReplayButton() {
    const replay = document.querySelector("#replay-btn");
    const lowerOrbit = document.querySelector("#lower-orbit");

    lowerOrbit.appendChild(replay);
    replay.style.marginTop = "20px";
    replay.style.textAlign = "center";
}

// ===============================
//  AVVIO COMPLETO
// ===============================

window.addEventListener("DOMContentLoaded", () => {
    startMedusaCycle();
    populateComfortZone();
    repositionReplayButton();
});
